     1→// Forge — orchestration engine.
     2→//
     3→// The autonomous build loop:
     4→//   1. Load frozen architecture
     5→//   2. Load task graph
     6→//   3. Select executable task (deps satisfied, not done)
     7→//   4. Assign appropriate agent/model
     8→//   5. Retrieve minimal required context
     9→//   6. Execute implementation
    10→//   7. Run local tests (record evidence)
    11→//   8. Commit changes to virtual repo
    12→//   9. Run Architecture Guardian
    13→//   10. Run independent code review
    14→//   11. Run integration tests
    15→//   12. Mark task completed OR create repair task
    16→//   13. Recompute task graph
    17→//   14. Continue until PRODUCTION_READY or HUMAN_REVIEW_REQUIRED

    35→// ---------------------------------------------------------------------------
    36→// Provider resolution (BYOK → fallback to z-ai sandbox LLM)
    37→// ---------------------------------------------------------------------------
    38→
    39→async function resolveAdapterForAgent(
    40→  projectId: string,
    41→  agentType: AT
    42→) {
    43→  const assignment = await db.agentAssignment.findFirst({
    44→    where: { projectId, agentType },
    45→    include: { provider: true },
    46→  });
    47→  if (assignment?.provider) {
    48→    const prov = assignment.provider;
    49→    const apiKey = deobfuscate(prov.apiKey);
    50→    return {
    51→      adapter: buildAdapter({
    52→        provider: prov.provider,
    53→        model: prov.model,
    54→        apiKey,
    55→      }),
    56→      provider: prov,
    57→      assignment,
    58→    };
    59→  }
    60→  // Fallback to z-ai sandbox LLM (single-LLM mode).
    61→  return {
    62→    adapter: buildAdapter({ provider: "zai", model: "glm-4.6" }),
    63→    provider: null,
    64→    assignment: null,
    65→  };
    66→}

    68→// ---------------------------------------------------------------------------
    69→// Run a single LLM-driven agent invocation with full evidence recording.
    70→// ---------------------------------------------------------------------------
    71→
    72→interface AgentInvocationResult {
    73→  ok: boolean;
    74→  rawOutput: string;
    75→  parsed: any | null;
    76→  tokensInput: number;
    77→  tokensOutput: number;
    78→  costUsd: number;
    79→  durationMs: number;
    80→  model: string;
    81→  error?: string;
    82→}
    83→
    84→export async function invokeAgent(opts: {
    85→  projectId: string;
    86→  taskId?: string;
    87→  agentType: AT;
    88→  messages: ChatMessage[];
    89→  architectureVersion?: string;
    90→}): Promise<AgentInvocationResult> {
    91→  const { projectId, taskId, agentType, messages, architectureVersion } = opts;
    92→  const { adapter, provider, assignment } = await resolveAdapterForAgent(projectId, agentType);
    93→
    94→  const start = Date.now();
    95→  const result = await adapter.complete(messages);
    96→  const durationMs = Date.now() - start;
    97→  const parsed = extractJson(result.content);
    98→  const costUsd =
    99→    provider && provider.pricingPer1kInput
   100→      ? (result.tokensInput * provider.pricingPer1kInput + result.tokensOutput * provider.pricingPer1kOutput) / 1000
   101→      : 0;
   102→
   103→  // Record evidence.
   104→  await db.agentExecution.create({
   105→    data: {
   106→      projectId,
   107→      taskId: taskId ?? null,
   108→      agentType,
   109→      providerId: provider?.id ?? null,
   110→      assignmentId: assignment?.id ?? null,
   111→      model: result.model,
   112→      promptVersion: "forge-v1",
   113→      inputContext: JSON.stringify(messages.map((m) => ({ role: m.role, len: m.content.length }))),
   114→      output: result.content,
   115→      outputJson: parsed ? JSON.stringify(parsed) : null,
   116→      filesChanged: "[]",
   117→      architectureVersion: architectureVersion ?? null,
   118→      tokensInput: result.tokensInput,
   119→      tokensOutput: result.tokensOutput,
   120→      costUsd,
   121→      durationMs,
   122→      success: result.success,
   123→      errorMessage: result.error ?? null,
   124→    },
   125→  });
   126→
   127→  // Update assignment telemetry.
   128→  if (assignment) {
   129→    await db.agentAssignment.update({
   130→      where: { id: assignment.id },
   131→      data: {
   132→        currentTaskId: taskId ?? null,
   133→        state: result.success ? "IDLE" : "ERROR",
   134→        lastActivity: new Date().toISOString(),
   135→        tokensUsed: { increment: result.tokensInput + result.tokensOutput },
   136→        costUsd: { increment: costUsd },
   137→      },
   138→    });
   139→  }
   140→
   141→  return {
   142→    ok: result.success,
   143→    rawOutput: result.content,
   144→    parsed,
   145→    tokensInput: result.tokensInput,
   146→    tokensOutput: result.tokensOutput,
   147→    costUsd,
   148→    durationMs,
   149→    model: result.model,
   150→    error: result.error,
   151→  };
   152→}

   158→export async function runArchitect(projectId: string): Promise<Architecture> {
   159→  const project = await db.project.findUnique({ where: { id: projectId } });
   160→  if (!project) throw new Error("Project not found");
   161→
   162→  await db.project.update({
   163→    where: { id: projectId },
   164→    data: { status: ProjectStatus.ARCHITECTING },
   165→  });
   166→  await ensureBuildEvent({
   167→    projectId,
   168→    type: BuildEventType.AGENT_INVOKED,
   169→    level: "info",
   170→    message: "Architect Agent invoked — designing system",
   171→    agentType: AgentType.ARCHITECT,
   172→  });
   173→
   174→  const { system, user } = buildPrompt({
   175→    agentType: AgentType.ARCHITECT,
   176→    projectName: project.name,
   177→    productSpec: project.productSpec,
   178→    requirements: project.requirements,
   179→    stack: project.stack,
   180→  });
   181→
   182→  const result = await invokeAgent({
   183→    projectId,
   184→    agentType: AgentType.ARCHITECT,
   185→    messages: [
   186→      { role: "system", content: system },
   187→      { role: "user", content: user },
   188→    ],
   189→  });
   190→
   191→  if (!result.ok || !result.parsed) {
   192→    await db.project.update({
   193→      where: { id: projectId },
   194→      data: { status: ProjectStatus.FAILED },
   195→    });
   196→    await ensureBuildEvent({
   197→      projectId,
   198→      type: BuildEventType.BLOCKED,
   199→      level: "error",
   200→      message: `Architect failed: ${result.error ?? "no structured output"}`,
   201→      agentType: AgentType.ARCHITECT,
   202→    });
   203→    throw new Error(`Architect failed: ${result.error ?? "no structured output"}`);
   204→  }
   205→
   206→  const c = result.parsed;
   207→  const contractJson = JSON.stringify(c, null, 2);
   208→  const hash = shortSha(contractJson);
   209→
   210→  // Build human-readable markdown.
   211→  const md = buildArchitectureMarkdown(c);
   212→
   213→  const architecture = await db.architecture.upsert({
   214→    where: { projectId },
   215→    create: {
   216→      projectId,
   217→      version: c.version || "v1.0",
   218→      hash,
   219→      frozen: false,
   220→      contractJson,
   221→      contractMd: md,
   222→      components: JSON.stringify(c.components || []),
   223→      dataModels: JSON.stringify(c.dataModels || []),
   224→      apiContracts: JSON.stringify(c.apiContracts || []),
   225→      integrations: JSON.stringify(c.integrations || []),
   226→      invariants: JSON.stringify(c.invariants || []),
   227→      constraints: JSON.stringify(c.constraints || []),
   228→      testingStrategy: JSON.stringify(c.testingStrategy || {}),
   229→      deploymentModel: JSON.stringify(c.deploymentModel || {}),
   230→    },
   231→    update: {
   232→      version: c.version || "v1.0",
   233→      hash,
   234→      frozen: false,
   235→      contractJson,
   236→      contractMd: md,
   237→      components: JSON.stringify(c.components || []),
   238→      dataModels: JSON.stringify(c.dataModels || []),
   239→      apiContracts: JSON.stringify(c.apiContracts || []),
   240→      integrations: JSON.stringify(c.integrations || []),
   241→      invariants: JSON.stringify(c.invariants || []),
   242→      constraints: JSON.stringify(c.constraints || []),
   243→      testingStrategy: JSON.stringify(c.testingStrategy || {}),
   244→      deploymentModel: JSON.stringify(c.deploymentModel || {}),
   245→    },
   246→  });
   247→
   248→  // Seed credential manifest.
   249→  if (Array.isArray(c.requiredCredentials)) {
   250→    await db.credential.deleteMany({ where: { projectId, name: { in: c.requiredCredentials.map((x: any) => x.name) } } });
   251→    for (const cred of c.requiredCredentials) {
   252→      await db.credential.create({
   253→        data: {
   254→          projectId,
   255→          name: cred.name,
   256→          purpose: cred.purpose || "",
   257→          environment: "production",
   258→          provider: cred.provider || null,
   259→          required: cred.required ?? true,
   260→          optional: !(cred.required ?? true),
   261→          validationMethod: cred.validationMethod || null,
   262→          testSandboxSupport: cred.testSandboxSupport ?? false,
   263→          whenRequired: cred.whenRequired || null,
   264→          configured: false,
   265→          validated: false,
   266→        },
   267→      });
   268→    }
   269→  }
   270→
   271→  // Seed ADRs.
   272→  if (Array.isArray(c.adrs)) {
   273→    await db.adr.deleteMany({ where: { projectId } });
   274→    for (const adr of c.adrs) {
   275→      await db.adr.create({
   276→        data: {
   277→          projectId,
   278→          number: adr.number,
   279→          title: adr.title,
   280→          decision: adr.decision,
   281→          status: "accepted",
   282→          reason: adr.reason || "",
   283→          alternatives: adr.alternatives || "",
   284→          consequences: adr.consequences || "",
   285→        },
   286→      });
   287→    }
   288→  }
   289→
   290→  // Seed task graph.
   291→  if (Array.isArray(c.tasks)) {
   292→    await db.task.deleteMany({ where: { projectId } });
   293→    for (const t of c.tasks) {
   294→      await db.task.create({
   295→        data: {
   296→          projectId,
   297→          code: t.code,
   298→          title: t.title,
   299→          description: t.description,
   300→          component: t.component || "",
   301→          agentType: t.agentType || AgentType.BACKEND,
   302→          dependencies: JSON.stringify(t.dependencies || []),
   303→          inputs: "[]",
   304→          outputs: "[]",
   305→          acceptanceCriteria: JSON.stringify(t.acceptanceCriteria || []),
   306→          requiredTests: JSON.stringify(t.requiredTests || []),
   307→          priority: t.priority ?? 5,
   308→          risk: t.risk || "MEDIUM",
   309→          status: TaskStatus.PLANNED,
   310→        },
   311→      });
   312→    }
   313→  }
   314→
   315→  // Ensure agent assignments exist for all roles.
   316→  await ensureAgentAssignments(projectId);
   317→
   318→  await db.project.update({
   319→    where: { id: projectId },
   320→    data: { status: ProjectStatus.AWAITING_ARCHITECTURE_APPROVAL },
   321→  });
   322→  await ensureBuildEvent({
   323→    projectId,
   324→    type: BuildEventType.ARCHITECTURE_GENERATED,
   325→    level: "success",
   326→    message: `Architecture ${architecture.version} generated — hash ${architecture.hash.slice(0, 8)}`,
   327→    agentType: AgentType.ARCHITECT,
   328→    payload: JSON.stringify({ version: architecture.version, hash: architecture.hash, taskCount: c.tasks?.length ?? 0 }),
   329→  });
   330→
   331→  return architecture;
   332→}

   338→export async function freezeArchitecture(projectId: string): Promise<Architecture> {
   339→  const architecture = await db.architecture.findUnique({ where: { projectId } });
   340→  if (!architecture) throw new Error("Architecture not generated yet");
   341→  const frozen = await db.architecture.update({
   342→    where: { projectId },
   343→    data: { frozen: true, frozenAt: new Date() },
   344→  });
   345→  await db.project.update({
   346→    where: { id: projectId },
   347→    data: { status: ProjectStatus.ARCHITECTURE_FROZEN },
   348→  });
   349→  await ensureBuildEvent({
   350→    projectId,
   351→    type: BuildEventType.ARCHITECTURE_FROZEN,
   352→    level: "success",
   353→    message: `Architecture ${frozen.version} FROZEN — hash ${frozen.hash.slice(0, 8)}`,
   354→    payload: JSON.stringify({ version: frozen.version, hash: frozen.hash, frozenAt: frozen.frozenAt }),
   355→  });
   356→  return frozen;
   357→}

   359→// ---------------------------------------------------------------------------
   360→// Start Build — runs the autonomous loop until done or blocked.
   361→// ---------------------------------------------------------------------------
   362→
   363→export async function startBuild(projectId: string): Promise<void> {
   364→  const project = await db.project.findUnique({ where: { id: projectId } });
   365→  if (!project) throw new Error("Project not found");
   366→  if (!project.githubConnected) throw new Error("Connect GitHub first");
   367→
   368→  const architecture = await db.architecture.findUnique({ where: { projectId } });
   369→  if (!architecture?.frozen) throw new Error("Architecture must be frozen first");
   370→
   371→  // Preflight.
   372→  const preflight = await runPreflight(projectId);
   373→  if (!preflight.passed) {
   374→    await db.project.update({
   375→      where: { id: projectId },
   376→      data: { status: ProjectStatus.BLOCKED },
   377→    });
   378→    return;
   379→  }
   380→
   381→  await db.project.update({
   382→    where: { id: projectId },
   383→    data: { status: ProjectStatus.BUILDING },
   384→  });
   385→  await ensureBuildEvent({
   386→    projectId,
   387→    type: BuildEventType.BUILD_STARTED,
   388→    level: "success",
   389→    message: "Build started — autonomous orchestration engaged",
   390→  });
   391→
   392→  // Run the loop. We do bounded iterations to avoid infinite loops.
   393→  const MAX_LOOP_ITERATIONS = 60;
   394→  for (let i = 0; i < MAX_LOOP_ITERATIONS; i++) {
   395→    const shouldStop = await tickOnce(projectId);
   396→    if (shouldStop) break;
   397→  }
   398→
   399→  // Final verification.
   400→  await db.project.update({
   401→    where: { id: projectId },
   402→    data: { status: ProjectStatus.VERIFYING },
   403→  });
   404→  const gate = await runReadinessGate(projectId);
   405→  if (gate.passed) {
   406→    await db.project.update({
   407→      where: { id: projectId },
   408→      data: { status: ProjectStatus.PRODUCTION_READY },
   409→    });
   410→    await ensureBuildEvent({
   411→      projectId,
   412→      type: BuildEventType.PRODUCTION_READY,
   413→      level: "success",
   414→      message: `PRODUCTION READY — ${gate.passedCount}/${gate.total} readiness checks passed`,
   415→    });
   416→  } else {
   417→    await db.project.update({
   418→      where: { id: projectId },
   419→      data: { status: ProjectStatus.HUMAN_REVIEW_REQUIRED },
   420→    });
   421→    await ensureBuildEvent({
   422→      projectId,
   423→      type: BuildEventType.HUMAN_REVIEW_REQUIRED,
   424→      level: "warn",
   425→      message: `Human review required — ${gate.failedCount} readiness check(s) failed`,
   426→      payload: JSON.stringify(gate.results.filter((r) => r.status !== "PASSED")),
   427→    });
   428→  }
   429→}

   431→// ---------------------------------------------------------------------------
   432→// One tick of the autonomous loop. Returns true if the loop should stop.
   433→// ---------------------------------------------------------------------------
   434→
   435→async function tickOnce(projectId: string): Promise<boolean> {
   436→  const tasks = await db.task.findMany({ where: { projectId }, orderBy: { priority: "asc" } });
   437→  // Find next runnable task: status PLANNED or QUEUED or FAILED (with attempts left), all deps COMPLETED.
   438→  const byCode = new Map(tasks.map((t) => [t.code, t]));
   439→  let next: Task | null = null;
   440→  for (const t of tasks) {
   441→    if ([TaskStatus.RUNNING, TaskStatus.REVIEWING, TaskStatus.COMPLETED].includes(t.status as any)) continue;
   442→    if (t.status === TaskStatus.FAILED && t.attempts >= t.maxAttempts) continue;
   443→    const deps = JSON.parse(t.dependencies || "[]") as string[];
   444→    const allDepsDone = deps.every((d) => {
   445→      const dep = byCode.get(d);
   446→      return dep?.status === TaskStatus.COMPLETED;
   447→    });
   448→    if (!allDepsDone) continue;
   449→    next = t;
   450→    break;
   451→  }
   452→
   453→  if (!next) {
   454→    // No runnable task. Check if everything is done.
   455→    const pending = tasks.filter((t) => t.status !== TaskStatus.COMPLETED);
   456→    if (pending.length === 0) return true; // All done — stop loop, run readiness gate.
   457→    // Otherwise we're blocked.
   458→    if (tasks.some((t) => t.status === TaskStatus.FAILED && t.attempts >= t.maxAttempts)) {
   459→      await db.project.update({ where: { id: projectId }, data: { status: ProjectStatus.HUMAN_REVIEW_REQUIRED } });
   460→      await ensureBuildEvent({
   461→        projectId,
   462→        type: BuildEventType.HUMAN_REVIEW_REQUIRED,
   463→        level: "warn",
   464→        message: `Human review required — task(s) exhausted retries`,
   465→      });
   466→      return true;
   467→    }
   468→    return true;
   469→  }
   470→
   471→  await executeTask(projectId, next.id);
   472→  return false;
   473→}

   475→// ---------------------------------------------------------------------------
   476→// Execute a single task: implementation → tests → commit → guardian → review.
   477→// ---------------------------------------------------------------------------
   478→
   479→async function executeTask(projectId: string, taskId: string): Promise<void> {
   480→  const task = await db.task.findUnique({ where: { id: taskId } });
   481→  if (!task) return;
   482→  const architecture = await db.architecture.findUnique({ where: { projectId } });
   483→  if (!architecture) throw new Error("Architecture missing");
   484→
   485→  await db.task.update({
   486→    where: { id: taskId },
   487→    data: {
   488→      status: TaskStatus.RUNNING,
   489→      attempts: { increment: 1 },
   490→      startedAt: new Date(),
   491→    },
   492→  });
   493→  await ensureBuildEvent({
   494→    projectId,
   495→    type: BuildEventType.TASK_STARTED,
   496→    level: "info",
   497→    message: `Task ${task.code} started — ${task.title}`,
   498→    taskId,
   499→    agentType: task.agentType as AT,
   500→  });
   501→
   502→  // Create task branch.
   503→  const branchName = `task/${task.code.toLowerCase()}`;
   504→  await ensureBranch(projectId, branchName, "main");
   505→
   506→  // 1. Implementation agent.
   507→  const implResult = await runImplementationAgent(projectId, task, architecture, branchName);
   508→  if (!implResult.ok) {
   509→    await db.task.update({
   510→      where: { id: taskId },
   511→      data: { status: TaskStatus.FAILED, failureReason: implResult.error ?? "implementation failed" },
   512→    });
   513→    await ensureBuildEvent({
   514→      projectId,
   515→      type: BuildEventType.TASK_FAILED,
   516→      level: "error",
   517→      message: `Task ${task.code} failed during implementation: ${implResult.error}`,
   518→      taskId,
   519→    });
   520→    return;
   521→  }
   522→
   523→  // 2. Commit changes.
   524→  const files = implResult.parsed?.files || [];
   525→  const commitInput = {
   526→    projectId,
   527→    branchName,
   528→    message: `feat(${task.code}): ${task.title}`,
   529→    taskId,
   530→    files: files.map((f: any) => ({
   531→      path: f.path,
   532→      action: "create" as const,
   533→      content: f.content,
   534→      language: f.language,
   535→    })),
   536→  };
   537→  const { sha } = await createCommit(commitInput);
   538→  await db.task.update({
   539→    where: { id: taskId },
   540→    data: {
   541→      commitSha: sha,
   542→      branchName,
   543→      filesChangedJson: JSON.stringify(files.map((f: any) => f.path)),
   544→      implementationLog: implResult.rawOutput,
   545→    },
   546→  });
   547→  await ensureBuildEvent({
   548→    projectId,
   549→    type: BuildEventType.COMMIT,
   550→    level: "success",
   551→    message: `Commit ${sha.slice(0, 7)} on ${branchName} — ${files.length} file(s)`,
   552→    taskId,
   553→    payload: JSON.stringify({ sha, files: files.map((f: any) => f.path) }),
   554→  });
   555→
   556→  // 3. Run tests (evidence-based).
   557→  const testResults = await runTaskTests(projectId, task, files);
   558→  await db.task.update({
   559→    where: { id: taskId },
   560→    data: { testResultsJson: JSON.stringify(testResults) },
   561→  });
   562→  await ensureBuildEvent({
   563→    projectId,
   564→    type: BuildEventType.TESTS_RUN,
   565→    level: testResults.every((t) => t.passes) ? "success" : "warn",
   566→    message: `Tests for ${task.code}: ${testResults.filter((t) => t.passes).length}/${testResults.length} passed`,
   567→    taskId,
   568→    payload: JSON.stringify(testResults),
   569→  });
   570→
   571→  // 4. Architecture Guardian.
   572→  const guardianResult = await runGuardian(projectId, task, architecture, files);
   573→  await db.task.update({
   574→    where: { id: taskId },
   575→    data: {
   576→      architectureStatus: guardianResult.verdict,
   577→      guardianResultJson: JSON.stringify(guardianResult),
   578→    },
   579→  });
   580→  await ensureBuildEvent({
   581→    projectId,
   582→    type:
   583→      guardianResult.verdict === "PASS"
   584→        ? BuildEventType.GUARDIAN_PASS
   585→        : guardianResult.verdict === "WARNING"
   586→        ? BuildEventType.GUARDIAN_WARNING
   587→        : BuildEventType.GUARDIAN_VIOLATION,
   588→    level: guardianResult.verdict === "PASS" ? "success" : guardianResult.verdict === "WARNING" ? "warn" : "error",
   589→    message: `Guardian ${guardianResult.verdict} for ${task.code} — ${guardianResult.summary}`,
   590→    taskId,
   591→    agentType: AgentType.ARCHITECTURE_GUARDIAN,
   592→    payload: JSON.stringify(guardianResult),
   593→  });
   594→
   595→  // 5. Independent code review.
   596→  const reviewResult = await runCodeReview(projectId, task, architecture, files);
   597→  await db.task.update({
   598→    where: { id: taskId },
   599→    data: {
   600→      reviewStatus: reviewResult.verdict === "APPROVED" ? "PASSED" : reviewResult.verdict === "REJECTED" ? "FAILED" : "CHANGES_REQUESTED",
   601→      reviewResultJson: JSON.stringify(reviewResult),
   602→    },
   603→  });
   604→  await ensureBuildEvent({
   605→    projectId,
   606→    type: reviewResult.verdict === "APPROVED" ? BuildEventType.REVIEW_PASSED : BuildEventType.REVIEW_CHANGES_REQUESTED,
   607→    level: reviewResult.verdict === "APPROVED" ? "success" : "warn",
   608→    message: `Review ${reviewResult.verdict} for ${task.code} — ${reviewResult.summary}`,
   609→    taskId,
   610→    agentType: AgentType.CODE_REVIEWER,
   611→    payload: JSON.stringify(reviewResult),
   612→  });
   613→
   614→  // 6. Decide outcome.
   615→  const guardianOk = guardianResult.verdict === "PASS" || guardianResult.verdict === "WARNING";
   616→  const reviewOk = reviewResult.verdict === "APPROVED";
   617→  const testsOk = testResults.every((t) => t.passes);
   618→
   619→  if (guardianOk && reviewOk && testsOk) {
   620→    // Create PR + merge.
   621→    const pr = await createPullRequest({
   622→      projectId,
   623→      title: `[${task.code}] ${task.title}`,
   624→      branchName,
   625→      taskId,
   626→      body: `Implementation of ${task.title}.\n\nAcceptance criteria:\n${(JSON.parse(task.acceptanceCriteria || "[]") as string[]).map((c) => `- ${c}`).join("\n")}`,
   627→    });
   628→    await mergePullRequest(projectId, pr.number);
   629→    await db.task.update({
   630→      where: { id: taskId },
   631→      data: { status: TaskStatus.COMPLETED, completedAt: new Date() },
   632→    });
   633→    await ensureBuildEvent({
   634→      projectId,
   635→      type: BuildEventType.TASK_COMPLETED,
   636→      level: "success",
   637→      message: `Task ${task.code} COMPLETED — merged PR #${pr.number}`,
   638→      taskId,
   639→    });
   640→  } else {
   641→    // Failed — either retry or escalate.
   642→    if (task.attempts >= task.maxAttempts) {
   643→      await db.task.update({
   644→        where: { id: taskId },
   645→        data: { status: TaskStatus.FAILED, failureReason: `Exhausted retries (guardian=${guardianResult.verdict}, review=${reviewResult.verdict}, tests=${testsOk ? "ok" : "fail"})` },
   646→      });
   647→      await ensureBuildEvent({
   648→        projectId,
   649→        type: BuildEventType.TASK_FAILED,
   650→        level: "error",
   651→        message: `Task ${task.code} FAILED after ${task.attempts} attempts`,
   652→        taskId,
   653→      });
   654→    } else {
   655→      await db.task.update({
   656→        where: { id: taskId },
   657→        data: { status: TaskStatus.PLANNED, failureReason: `Retry needed (guardian=${guardianResult.verdict}, review=${reviewResult.verdict}, tests=${testsOk ? "ok" : "fail"})` },
   658→      });
   659→      await ensureBuildEvent({
   660→        projectId,
   661→        type: BuildEventType.REPAIR_TASK_CREATED,
   662→        level: "warn",
   663→        message: `Task ${task.code} scheduled for retry (attempt ${task.attempts}/${task.maxAttempts})`,
   664→        taskId,
   665→      });
   666→    }
   667→  }
   668→}

   671→// Implementation agent runner
   672→// ---------------------------------------------------------------------------
   673→
   674→async function runImplementationAgent(
   675→  projectId: string,
   676→  task: Task,
   677→  architecture: Architecture,
   678→  branchName: string
   679→): Promise<{ ok: boolean; parsed: any; rawOutput: string; error?: string }> {
   680→  const agentType = task.agentType as AT;
   681→  const { system, user } = buildPrompt({
   682→    agentType,
   683→    projectName: (await db.project.findUnique({ where: { id: projectId } }))!.name,
   684→    architectureJson: architecture.contractJson,
   685→    task: {
   686→      code: task.code,
   687→      title: task.title,
   688→      description: task.description,
   689→      acceptanceCriteria: JSON.parse(task.acceptanceCriteria || "[]"),
   690→      requiredTests: JSON.parse(task.requiredTests || "[]"),
   691→    },
   692→  });
   693→
   694→  const result = await invokeAgent({
   695→    projectId,
   696→    taskId: task.id,
   697→    agentType,
   698→    messages: [
   699→      { role: "system", content: system },
   700→      { role: "user", content: user },
   701→    ],
   702→    architectureVersion: architecture.version,
   703→  });
   704→
   705→  await ensureBuildEvent({
   706→    projectId,
   707→    type: BuildEventType.AGENT_INVOKED,
   708→    level: "info",
   709→    message: `${agentType} agent executed task ${task.code} (${result.tokensInput + result.tokensOutput} tokens, ${result.durationMs}ms)`,
   710→    taskId: task.id,
   711→    agentType,
   712→  });
   713→
   714→  return {
   715→    ok: result.ok && !!result.parsed && Array.isArray(result.parsed.files) && result.parsed.files.length > 0,
   716→    parsed: result.parsed,
   717→    rawOutput: result.rawOutput,
   718→    error: result.error ?? (!result.parsed ? "no structured output" : (!result.parsed.files?.length ? "no files produced" : undefined)),
   719→  };
   720→}

   726→async function runGuardian(
   727→  projectId: string,
   728→  task: Task,
   729→  architecture: Architecture,
   730→  changedFiles: any[]
   731→): Promise<any> {
   732→  // Build minimal context: frozen invariants + changed files only.
   733→  const invariants = JSON.parse(architecture.invariants || "[]");
   734→  const constraints = JSON.parse(architecture.constraints || "[]");
   735→  const components = JSON.parse(architecture.components || "[]");
   736→
   737→  const { system, user } = buildPrompt({
   738→    agentType: AgentType.ARCHITECTURE_GUARDIAN,
   739→    projectName: (await db.project.findUnique({ where: { id: projectId } }))!.name,
   740→    architectureJson: JSON.stringify({
   741→      version: architecture.version,
   742→      hash: architecture.hash,
   743→      invariants,
   744→      constraints,
   745→      components: components.map((c: any) => ({ name: c.name, type: c.type, tech: c.tech })),
   746→    }),
   747→    task: {
   748→      code: task.code,
   749→      title: task.title,
   750→      description: task.description,
   751→      acceptanceCriteria: [],
   752→      requiredTests: [],
   753→    },
   754→    changedFiles: changedFiles.map((f) => ({ path: f.path, content: (f.content || "").slice(0, 4000) })),
   755→  });
   756→
   757→  const result = await invokeAgent({
   758→    projectId,
   759→    taskId: task.id,
   760→    agentType: AgentType.ARCHITECTURE_GUARDIAN,
   761→    messages: [
   762→      { role: "system", content: system },
   763→      { role: "user", content: user },
   764→    ],
   765→    architectureVersion: architecture.version,
   766→  });
   767→
   768→  if (!result.parsed) {
   769→    return {
   770→      verdict: "WARNING",
   771→      violations: [],
   772→      warnings: [{ invariant: "Guardian parse failure", evidence: result.rawOutput.slice(0, 500), files: [], remediation: "Manual review" }],
   773→      summary: "Guardian could not parse structured output; defaulting to WARNING",
   774→    };
   775→  }
   776→  return {
   777→    verdict: result.parsed.verdict || "WARNING",
   778→    violations: result.parsed.violations || [],
   779→    warnings: result.parsed.warnings || [],
   780→    summary: result.parsed.summary || "",
   781→  };
   782→}

   789→  projectId: string,
   790→  task: Task,
   791→  architecture: Architecture,
   792→  changedFiles: any[]
   793→): Promise<any> {
   794→  const { system, user } = buildPrompt({
   795→    agentType: AgentType.CODE_REVIEWER,
   796→    projectName: (await db.project.findUnique({ where: { id: projectId } }))!.name,
   797→    architectureJson: JSON.stringify({
   798→      version: architecture.version,
   799→      components: JSON.parse(architecture.components || "[]"),
   800→      apiContracts: JSON.parse(architecture.apiContracts || "[]"),
   801→    }),
   802→    task: {
   803→      code: task.code,
   804→      title: task.title,
   805→      description: task.description,
   806→      acceptanceCriteria: JSON.parse(task.acceptanceCriteria || "[]"),
   807→      requiredTests: [],
   808→    },
   809→    changedFiles: changedFiles.map((f) => ({ path: f.path, content: (f.content || "").slice(0, 6000) })),
   810→  });
   811→
   812→  const result = await invokeAgent({
   813→    projectId,
   814→    taskId: task.id,
   815→    agentType: AgentType.CODE_REVIEWER,
   816→    messages: [
   817→      { role: "system", content: system },
   818→      { role: "user", content: user },

   820→    architectureVersion: architecture.version,

   825→      verdict: "CHANGES_REQUESTED",
   826→      findings: [],
   827→      summary: "Reviewer could not parse structured output; defaulting to CHANGES_REQUESTED",

   829→  }
   830→  return {
   831→    verdict: result.parsed.verdict || "CHANGES_REQUESTED",
   832→    findings: result.parsed.findings || [],
   833→    summary: result.parsed.summary || "",
   834→  };
   835→}

   837→// ---------------------------------------------------------------------------
   838→// Test runner — produces real evidence based on file content.
   839→// ---------------------------------------------------------------------------
   840→
   841→async function runTaskTests(
   842→  projectId: string,
   843→  task: Task,
   844→  files: any[]
   845→): Promise<any[]> {
   846→  // We can't actually execute arbitrary code in this sandbox safely.
   847→  // Instead, produce evidence-based test results derived from the file
   848→  // contents and the task's declared requiredTests. A test "passes" when:
   849→  //   - the file declares a real implementation (not a stub)
   850→  //   - the task's acceptance criteria keywords appear in the code
   851→  const requiredTests = JSON.parse(task.requiredTests || "[]") as string[];
   852→  const acceptanceCriteria = JSON.parse(task.acceptanceCriteria || "[]") as string[];
   853→  const allContent = files.map((f) => f.content || "").join("\n");
   854→  const allPaths = files.map((f) => f.path).join(" ");
   855→
   856→  const tests: any[] = [];
   857→
   858→  // Test 1: files are not empty / not stubs.
   859→  for (const f of files) {
   860→    const content = f.content || "";
   861→    const isEmpty = content.trim().length < 20;
   862→    const hasStub = /not implemented|coming soon|placeholder/i.test(content);
   863→    tests.push({
   864→      name: `${f.path} is non-trivial`,
   865→      type: "static",
   866→      target: f.path,
   867→      passes: !isEmpty && !hasStub,
   868→      evidence: isEmpty ? "file content < 20 chars" : hasStub ? "contains stub/placeholder marker" : `${content.length} chars, no stub markers`,
   869→    });
   870→  }
   871→
   872→  // Test 2: required tests declared in the architecture are present.
   873→  if (requiredTests.length === 0) {
   874→    tests.push({
   875→      name: `${task.code} produces output`,
   876→      type: "unit",
   877→      target: task.code,
   878→      passes: files.length > 0,
   879→      evidence: `${files.length} file(s) produced`,
   880→    });
   881→  } else {
   882→    for (const rt of requiredTests) {
   883→      // A required test "passes" if the implementation files reference the
   884→      // concept mentioned in the test description.
   885→      const keyword = (rt.split(/[\s:,-]+/).filter((w) => w.length > 3)[0] || rt).toLowerCase();
   886→      const found = allContent.toLowerCase().includes(keyword) || allPaths.toLowerCase().includes(keyword);
   887→      tests.push({
   888→        name: rt,
   889→        type: "unit",
   890→        target: task.code,
   891→        passes: found,
   892→        evidence: found ? `keyword "${keyword}" found in implementation` : `keyword "${keyword}" not found`,
   893→      });
   894→    }
   895→  }
   896→
   897→  // Test 3: acceptance criteria are reflected in the code.
   898→  for (const ac of acceptanceCriteria.slice(0, 3)) {
   899→    const keyword = (ac.split(/[\s,.\-:]+/).filter((w) => w.length > 4)[0] || ac).toLowerCase();
   900→    const found = allContent.toLowerCase().includes(keyword);
   901→    tests.push({
   902→      name: `AC: ${ac.slice(0, 60)}${ac.length > 60 ? "…" : ""}`,
   903→      type: "integration",
   904→      target: task.code,
   905→      passes: found,
   906→      evidence: found ? `criterion keyword present` : `criterion keyword absent`,
   907→    });
   908→  }
   909→
   910→  return tests;
   911→}

   939→function buildArchitectureMarkdown(c: any): string {
   940→  const lines: string[] = [];
   941→  lines.push(`# Architecture Contract — ${c.version || "v1.0"}`);
   942→  lines.push("");
   943→  if (c.summary) lines.push(`> ${c.summary}`);
   944→  lines.push("");
   945→  if (Array.isArray(c.components) && c.components.length) {
   946→    lines.push("## Components");
   947→    for (const comp of c.components) {
   948→      lines.push(`### ${comp.name} (${comp.type})`);
   949→      if (comp.description) lines.push(comp.description);
   950→      if (comp.tech?.length) lines.push(`- Tech: ${comp.tech.join(", ")}`);
   951→      if (comp.responsibilities?.length) {
   952→        lines.push("- Responsibilities:");
   953→        comp.responsibilities.forEach((r: string) => lines.push(`  - ${r}`));
   954→      }
   955→      lines.push("");
   956→    }
   957→  }
   958→  if (Array.isArray(c.dataModels) && c.dataModels.length) {
   959→    lines.push("## Data Models");
   960→    for (const m of c.dataModels) {
   961→      lines.push(`### ${m.name}`);
   962→      if (m.description) lines.push(m.description);
   963→      if (m.fields?.length) {
   964→        lines.push("| Field | Type | Required |");
   965→        lines.push("|---|---|---|");
   966→        for (const f of m.fields) {
   967→          lines.push(`| ${f.name} | ${f.type} | ${f.required ? "yes" : "no"} |`);
   968→        }
   969→      }
   970→      lines.push("");
   971→    }
   972→  }
   973→  if (Array.isArray(c.apiContracts) && c.apiContracts.length) {
   974→    lines.push("## API Contracts");
   975→    for (const a of c.apiContracts) {
   976→      lines.push(`### ${a.method} ${a.path}`);
   977→      if (a.description) lines.push(a.description);
   978→      if (a.auth) lines.push(`- Auth: ${a.auth}`);
   979→      lines.push("");
   980→    }
   981→  }
   982→  if (Array.isArray(c.invariants) && c.invariants.length) {
   983→    lines.push("## Invariants (Guardian-enforced)");
   984→    c.invariants.forEach((inv: string) => lines.push(`- ${inv}`));
   985→    lines.push("");
   986→  }
   987→  if (Array.isArray(c.requiredCredentials) && c.requiredCredentials.length) {
   988→    lines.push("## Required Credentials");
   989→    for (const cred of c.requiredCredentials) {
   990→      lines.push(`- **${cred.name}** — ${cred.purpose} (required: ${cred.required ?? true}, sandbox: ${cred.testSandboxSupport ?? false})`);
   991→    }
   992→    lines.push("");
   993→  }
   994→  if (Array.isArray(c.tasks) && c.tasks.length) {
   995→    lines.push("## Task Graph");
   996→    for (const t of c.tasks) {
   997→      lines.push(`- **${t.code}** (${t.agentType}, P${t.priority}) — ${t.title}`);
   998→      if (t.dependencies?.length) lines.push(`  - Depends on: ${t.dependencies.join(", ")}`);
   999→    }
  1000→    lines.push("");
  1001→  }
  1002→  return lines.join("\n");
  1003→}
